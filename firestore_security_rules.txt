rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    
    // Helper functions
    function isAuthenticated() {
      return request.auth.uid != null;
    }
    
    function isUserOwner(userId) {
      return request.auth.uid == userId;
    }
    
    function isValidBloggerData() {
      let hasRequiredFields = request.resource.data.keys().hasAll(['uid', 'email', 'displayName']);
      let noForbiddenUpdates = !request.resource.data.keys().hasAny(['uid', 'email', 'createdAt']);
      return hasRequiredFields && noForbiddenUpdates;
    }
    
    // Users Collection - Blogger Profiles
    match /users/{userId} {
      // Allow anyone to read public blogger profiles
      allow read: if isAuthenticated();
      
      // Allow users to create their own profile on signup
      allow create: if isAuthenticated() && 
                       isUserOwner(userId) && 
                       isValidBloggerData() &&
                       request.resource.data.verificationStatus == 'Approved';
      
      // Allow users to update only their own profile
      allow update: if isAuthenticated() && 
                       isUserOwner(userId) && 
                       isValidBloggerData();
      
      // Prevent deletion of profiles (soft delete only via custom claim or admin)
      allow delete: if false;
    }

    // Blogs Collection
    match /blogs/{blogId} {
      // Allow authenticated users to read all blogs
      allow read: if isAuthenticated();

      // Allow authenticated users to create blog entries
      allow create: if isAuthenticated() &&
                       request.resource.data.keys().hasAll([
                         'title', 'domainLink', 'uploadedAt', 'uploadedBy'
                       ]) &&
                       request.resource.data.uploadedBy == request.auth.uid;

      // Allow authors to update their own blog entries
      allow update: if isAuthenticated() &&
                       resource.data.uploadedBy == request.auth.uid;

      // Allow authors to delete their own blog entries
      allow delete: if isAuthenticated() &&
                       resource.data.uploadedBy == request.auth.uid;
    }
    
    // Future: Ratings Collection (when implemented)
    match /ratings/{ratingId} {
      // Allow authenticated users to read all ratings
      allow read: if isAuthenticated();
      
      // Allow authenticated users to create ratings
      allow create: if isAuthenticated() && 
                       request.resource.data.raterId == request.auth.uid &&
                       request.resource.data.rating >= 1 &&
                       request.resource.data.rating <= 5;
      
      // Allow only the rater to update their own rating
      allow update: if isAuthenticated() && 
                       resource.data.raterId == request.auth.uid;
      
      // Allow only the rater to delete their own rating
      allow delete: if isAuthenticated() && 
                       resource.data.raterId == request.auth.uid;
    }
    
    // Future: Messages Collection (when implemented)
    match /messages/{messageId} {
      // Allow users to read their own messages
      allow read: if isAuthenticated() && 
                      (request.auth.uid == resource.data.senderId || 
                       request.auth.uid == resource.data.recipientId);
      
      // Allow authenticated users to create messages
      allow create: if isAuthenticated() && 
                       request.resource.data.senderId == request.auth.uid;
      
      // Prevent updates and deletes on messages
      allow update, delete: if false;
    }
    
    // Catch-all: deny everything else
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
